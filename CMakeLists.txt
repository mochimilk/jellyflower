cmake_minimum_required(VERSION 3.24)
project(jellyflower VERSION 0.23.0 LANGUAGES C CXX RC)

if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Build on Windows with Visual Studio 2022 C++ tools and a Windows SDK (20348 or newer).")
endif()
if(CMAKE_VS_WINDOWS_TARGET_PLATFORM_VERSION AND CMAKE_VS_WINDOWS_TARGET_PLATFORM_VERSION VERSION_LESS "10.0.20348.0")
  message(FATAL_ERROR "Windows SDK 10.0.20348 or newer is required for the Windows first-name API.")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Use x64 or ARM64, matching Windows Explorer. x86 is not supported.")
endif()

set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:/utf-8>")
include(cmake/Jellyflower.cmake)
configure_file(src/version.hpp.in version.hpp @ONLY)
configure_file(src/version.rc.in version.rc @ONLY)
add_library(jellyflower_zip STATIC third_party/miniz/miniz.c third_party/miniz/miniz_tinfl.c third_party/miniz/miniz_zip.c)
target_include_directories(jellyflower_zip PUBLIC third_party/miniz)
target_compile_definitions(jellyflower_zip PUBLIC MINIZ_NO_DEFLATE_APIS MINIZ_NO_ZLIB_APIS MINIZ_NO_ZLIB_COMPATIBLE_NAMES)
target_compile_options(jellyflower_zip PRIVATE /W0)
add_library(jellyflower_cursor_pack STATIC src/cursor_pack.cpp)
target_compile_features(jellyflower_cursor_pack PRIVATE cxx_std_17)
target_compile_definitions(jellyflower_cursor_pack PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(jellyflower_cursor_pack PRIVATE /W4 /permissive- /EHsc /utf-8)
target_link_libraries(jellyflower_cursor_pack PUBLIC jellyflower_zip user32 shell32 ole32)
add_executable(AcrylicTaskbar WIN32 src/main.cpp src/taskbar_slot.cpp src/startup.cpp src/branding.cpp src/updates.cpp src/update_download.cpp src/auto_theme.cpp src/clock_options.cpp src/taskbar_probe.cpp src/tray.cpp src/popup.cpp src/menu_glass.cpp src/theme.cpp src/cursors.cpp src/cursor_icon.cpp src/bridge_notify.cpp src/keyboard_field.cpp src/compact_screen.cpp src/resources.rc src/app.manifest)
set_target_properties(AcrylicTaskbar PROPERTIES OUTPUT_NAME jellyflower)
target_include_directories(AcrylicTaskbar PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_include_directories(AcrylicTaskbar PRIVATE assets)
add_library(AcrylicTaskbarBridge SHARED src/bridge.cpp src/adaptive_contrast.cpp src/adaptive_session.cpp src/clock.cpp src/clock_options.cpp src/volume.cpp src/chevron.cpp src/tray_only.cpp src/dock.cpp src/bar_height.cpp src/bridge_notify.cpp src/bridge.def)
add_library(JellyflowerStart SHARED src/start_bridge.cpp src/start_bridge.def)
set_target_properties(JellyflowerStart PROPERTIES OUTPUT_NAME jellyflower-start-v21)
target_compile_features(JellyflowerStart PRIVATE cxx_std_17)
target_compile_definitions(JellyflowerStart PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_compile_options(JellyflowerStart PRIVATE /W4 /permissive- /EHsc /utf-8)
target_link_libraries(JellyflowerStart PRIVATE windowsapp user32 ole32 oleaut32 uuid dxguid)
set_target_properties(AcrylicTaskbarBridge PROPERTIES OUTPUT_NAME jellyflower-bridge-v20)
foreach(target AcrylicTaskbar AcrylicTaskbarBridge)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_compile_definitions(${target} PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN
    _WIN32_WINNT=0x0A00 WINVER=0x0A00)
  target_compile_options(${target} PRIVATE /W4 /permissive- /EHsc /utf-8)
  target_link_libraries(${target} PRIVATE user32 ole32 oleaut32 shell32 advapi32 uuid gdi32)
endforeach()
target_link_libraries(AcrylicTaskbarBridge PRIVATE windowsapp dxguid windowscodecs shcore wtsapi32 dwmapi)
target_compile_features(AcrylicTaskbar PRIVATE cxx_std_20)
target_link_libraries(AcrylicTaskbar PRIVATE uiautomationcore taskschd dwmapi gdi32 gdiplus d2d1 d3d11 dxguid dwrite comctl32 msimg32 winhttp bcrypt windowsapp CoreMessaging jellyflower_cursor_pack)
target_sources(AcrylicTaskbar PRIVATE src/search_identity.cpp)
target_link_libraries(AcrylicTaskbar PRIVATE netapi32 shcore dbghelp)
target_link_options(AcrylicTaskbar PRIVATE /MANIFEST:EMBED)
target_sources(AcrylicTaskbar PRIVATE src/start_session.cpp)
target_sources(AcrylicTaskbar PRIVATE src/wallpaper.cpp src/wallpaper_state.cpp)
add_dependencies(AcrylicTaskbar AcrylicTaskbarBridge JellyflowerStart)

if(CMAKE_GENERATOR_PLATFORM STREQUAL "ARM64")
  set(JELLYFLOWER_ARCH ARM64)
else()
  set(JELLYFLOWER_ARCH x64)
endif()
set(payload "${CMAKE_CURRENT_BINARY_DIR}/payload-$<CONFIG>.rc")
add_custom_command(OUTPUT "${payload}"
  COMMAND "${CMAKE_COMMAND}" "-DAPP=$<TARGET_FILE:AcrylicTaskbar>" "-DBRIDGE=$<TARGET_FILE:AcrylicTaskbarBridge>"
    "-DSTART=$<TARGET_FILE:JellyflowerStart>" "-DROOT=${CMAKE_CURRENT_SOURCE_DIR}" "-DOUTPUT=${payload}" -P "${CMAKE_CURRENT_SOURCE_DIR}/tools/WritePayload.cmake"
  DEPENDS AcrylicTaskbar AcrylicTaskbarBridge JellyflowerStart tools/WritePayload.cmake README.md LICENSE.md THIRD_PARTY_NOTICES.md
    assets/LICENSE-LUCIDE.txt assets/app-dark.ico assets/app-light.ico VERBATIM)
add_executable(JellyflowerSetup WIN32 src/setup.cpp src/startup.cpp src/setup_resources.rc src/app.manifest "${payload}")
set_target_properties(JellyflowerSetup PROPERTIES OUTPUT_NAME "jellyflower-setup-${JELLYFLOWER_ARCH}")
target_include_directories(JellyflowerSetup PRIVATE assets "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_features(JellyflowerSetup PRIVATE cxx_std_17)
target_compile_definitions(JellyflowerSetup PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_compile_options(JellyflowerSetup PRIVATE /W4 /permissive- /EHsc /utf-8)
target_link_libraries(JellyflowerSetup PRIVATE user32 shell32 ole32 oleaut32 uuid advapi32 taskschd)
target_link_options(JellyflowerSetup PRIVATE /MANIFEST:EMBED)

install(TARGETS AcrylicTaskbar AcrylicTaskbarBridge JellyflowerStart JellyflowerSetup RUNTIME DESTINATION .)
install(FILES assets/app-dark.ico assets/app-light.ico DESTINATION .)
install(FILES scripts/Install.ps1 scripts/Uninstall.ps1 scripts/Diagnose.ps1
  README.md LICENSE.md THIRD_PARTY_NOTICES.md DESTINATION .)
install(DIRECTORY docs DESTINATION .)
install(FILES assets/LICENSE-LUCIDE.txt DESTINATION .)
install(FILES assets/LICENSE-OCTICONS.txt DESTINATION .)

option(ACRYLIC_BUILD_TESTS "Build native layout, shell lifetime, cursor, theme, and bridge checks" OFF)
if(ACRYLIC_BUILD_TESTS)
  enable_testing()
  include(tests/Tests.cmake)
endif()
