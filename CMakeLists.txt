cmake_minimum_required(VERSION 3.24)
project(jellyflower VERSION 0.13.0 LANGUAGES CXX RC)

if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Build on Windows with Visual Studio 2022 C++ tools and a Windows SDK (19041 or newer).")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Use x64 or ARM64, matching Windows Explorer. x86 is not supported.")
endif()

set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
configure_file(src/version.hpp.in version.hpp @ONLY)
configure_file(src/version.rc.in version.rc @ONLY)
add_executable(AcrylicTaskbar WIN32 src/main.cpp src/branding.cpp src/updates.cpp src/clock_options.cpp src/clock_font.cpp src/taskbar_probe.cpp src/tray.cpp src/popup.cpp src/theme.cpp src/cursors.cpp src/cursor_icon.cpp src/bridge_notify.cpp src/resources.rc src/app.manifest)
set_target_properties(AcrylicTaskbar PROPERTIES OUTPUT_NAME jellyflower)
target_include_directories(AcrylicTaskbar PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_include_directories(AcrylicTaskbar PRIVATE assets)
add_library(AcrylicTaskbarBridge SHARED src/bridge.cpp src/clock.cpp src/clock_options.cpp src/clock_font.cpp src/volume.cpp src/chevron.cpp src/bridge_notify.cpp src/bridge.def)
set_target_properties(AcrylicTaskbarBridge PROPERTIES OUTPUT_NAME jellyflower-bridge)
foreach(target AcrylicTaskbar AcrylicTaskbarBridge)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_compile_definitions(${target} PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN
    _WIN32_WINNT=0x0A00 WINVER=0x0A00)
  target_compile_options(${target} PRIVATE /W4 /permissive- /EHsc /utf-8)
  target_link_libraries(${target} PRIVATE user32 ole32 oleaut32 shell32 advapi32 uuid dwrite gdi32)
endforeach()
target_link_libraries(AcrylicTaskbarBridge PRIVATE windowsapp)
target_link_libraries(AcrylicTaskbar PRIVATE dwmapi gdi32 comctl32 msimg32 winhttp windowsapp)
target_link_options(AcrylicTaskbar PRIVATE /MANIFEST:EMBED)
add_dependencies(AcrylicTaskbar AcrylicTaskbarBridge)

if(CMAKE_GENERATOR_PLATFORM STREQUAL "ARM64")
  set(JELLYFLOWER_ARCH ARM64)
else()
  set(JELLYFLOWER_ARCH x64)
endif()
set(payload "${CMAKE_CURRENT_BINARY_DIR}/payload-$<CONFIG>.rc")
add_custom_command(OUTPUT "${payload}"
  COMMAND "${CMAKE_COMMAND}" "-DAPP=$<TARGET_FILE:AcrylicTaskbar>" "-DBRIDGE=$<TARGET_FILE:AcrylicTaskbarBridge>"
    "-DROOT=${CMAKE_CURRENT_SOURCE_DIR}" "-DOUTPUT=${payload}" -P "${CMAKE_CURRENT_SOURCE_DIR}/tools/WritePayload.cmake"
  DEPENDS AcrylicTaskbar AcrylicTaskbarBridge tools/WritePayload.cmake README.md LICENSE.md THIRD_PARTY_NOTICES.md
    assets/LICENSE-LUCIDE.txt assets/app-dark.ico assets/app-light.ico VERBATIM)
add_executable(JellyflowerSetup WIN32 src/setup.cpp src/setup_resources.rc src/app.manifest "${payload}")
set_target_properties(JellyflowerSetup PROPERTIES OUTPUT_NAME "jellyflower-setup-${JELLYFLOWER_ARCH}")
target_include_directories(JellyflowerSetup PRIVATE assets "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_features(JellyflowerSetup PRIVATE cxx_std_17)
target_compile_definitions(JellyflowerSetup PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_compile_options(JellyflowerSetup PRIVATE /W4 /permissive- /EHsc /utf-8)
target_link_libraries(JellyflowerSetup PRIVATE user32 shell32 ole32 uuid advapi32)
target_link_options(JellyflowerSetup PRIVATE /MANIFEST:EMBED)

install(TARGETS AcrylicTaskbar AcrylicTaskbarBridge JellyflowerSetup RUNTIME DESTINATION .)
install(FILES assets/app-dark.ico assets/app-light.ico DESTINATION .)
install(FILES scripts/Install.ps1 scripts/Uninstall.ps1 scripts/Diagnose.ps1
  README.md LICENSE.md THIRD_PARTY_NOTICES.md DESTINATION .)
install(DIRECTORY docs DESTINATION .)
install(FILES assets/LICENSE-LUCIDE.txt DESTINATION .)
